package com.mycompany.studentjdbc;

import java.sql.Connection;
import java.sql.DriverManager;

public class DBConnection {

    public static Connection getConnection() {

        try {
            String url = "jdbc:mysql://localhost:3306/studentdb";
            String user = "root";
            String password = "root";

            Connection con = DriverManager.getConnection(
                    url, user, password);

            System.out.println("Database Connected!");

            return con;

        } catch (Exception e) {
            System.out.println(e.getMessage());
            return null;
        }
    }
}

StudentCrud.java
package com.mycompany.studentjdbc;

import com.mycompany.studentjdbc.DBConnection;
import java.sql.*;
import java.util.Scanner;

public class StudentCRUD {

public static void main(String[] args) {

    Scanner sc = new Scanner(System.in);

    try {
        Connection con = DBConnection.getConnection();

        if (con == null) {
            System.out.println("Database Connection Failed!");
            return;
        }

        int choice;

        do {
            System.out.println("\n===== STUDENT DATABASE =====");
            System.out.println("1. Insert Student");
            System.out.println("2. Display Students");
            System.out.println("3. Update Student");
            System.out.println("4. Delete Student");
            System.out.println("5. Exit");

            System.out.print("Enter Choice: ");
            choice = sc.nextInt();
            sc.nextLine();

            switch (choice) {

                case 1:
                    System.out.print("Enter Roll No: ");
                    int roll = sc.nextInt();
                    sc.nextLine();

                    System.out.print("Enter Name: ");
                    String name = sc.nextLine();

                    System.out.print("Enter Department: ");
                    String dept = sc.nextLine();

                    System.out.print("Enter Marks: ");
                    double marks = sc.nextDouble();

                    String insert = "INSERT INTO students VALUES(?,?,?,?)";

                    PreparedStatement ps1 = con.prepareStatement(insert);
                    ps1.setInt(1, roll);
                    ps1.setString(2, name);
                    ps1.setString(3, dept);
                    ps1.setDouble(4, marks);

                    ps1.executeUpdate();

                    System.out.println("Student Inserted Successfully!");
                    ps1.close();
                    break;

                case 2:
                    String select = "SELECT * FROM students";

                    Statement st = con.createStatement();
                    ResultSet rs = st.executeQuery(select);

                    System.out.println("\n----- Student Records -----");

                    while (rs.next()) {
                        System.out.println(
                            rs.getInt("roll_no") + " | " +
                            rs.getString("name") + " | " +
                            rs.getString("department") + " | " +
                            rs.getDouble("marks")
                        );
                    }

                    rs.close();
                    st.close();
                    break;

                case 3:
                    System.out.print("Enter Roll No to Update: ");
                    int updateRoll = sc.nextInt();
                    sc.nextLine();

                    System.out.print("Enter New Name: ");
                    String newName = sc.nextLine();

                    System.out.print("Enter New Department: ");
                    String newDept = sc.nextLine();

                    System.out.print("Enter New Marks: ");
                    double newMarks = sc.nextDouble();

                    String update = "UPDATE students SET name=?, department=?, marks=? WHERE roll_no=?";

                    PreparedStatement ps2 = con.prepareStatement(update);
                    ps2.setString(1, newName);
                    ps2.setString(2, newDept);
                    ps2.setDouble(3, newMarks);
                    ps2.setInt(4, updateRoll);

                    int updated = ps2.executeUpdate();

                    if (updated > 0)
                        System.out.println("Student Updated Successfully!");
                    else
                        System.out.println("Student Not Found!");

                    ps2.close();
                    break;

                case 4:
                    System.out.print("Enter Roll No to Delete: ");
                    int deleteRoll = sc.nextInt();

                    String delete = "DELETE FROM students WHERE roll_no=?";

                    PreparedStatement ps3 = con.prepareStatement(delete);
                    ps3.setInt(1, deleteRoll);

                    int deleted = ps3.executeUpdate();

                    if (deleted > 0)
                        System.out.println("Student Deleted Successfully!");
                    else
                        System.out.println("Student Not Found!");

                    ps3.close();
                    break;

                case 5:
                    System.out.println("Exiting Program...");
                    break;

                default:
                    System.out.println("Invalid Choice!");
            }

        } while (choice != 5);

        con.close();

    } catch (Exception e) {
        System.out.println("Error: " + e.getMessage());
    }

    sc.close();
}

}

main.java
package com.mycompany.studentjdbc;

import java.sql.Connection;

public class Main {

public static void main(String[] args) {

    Connection con = DBConnection.getConnection();

    if (con != null) {
        System.out.println("JDBC Connection Successful!");
        
        try {
            con.close();
            System.out.println("Database Connection Closed!");
        } catch (Exception e) {
            System.out.println(e);
        }

    } else {
        System.out.println("JDBC Connection Failed!");
    }
}

}
